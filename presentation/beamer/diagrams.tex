% --- a screenshot with its caption, side by side ---------------------------

% #1 asset name (in assets/), #2 image width, #3 caption text
%
% Both dimensions are capped. The screenshots are all 960x570, so a width that
% fills the column would stand taller than the frame once the caption is
% beside it; the caption is set small for the same reason.
\newcommand{\columnsimage}[3]{%
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.63\textwidth}
      \includegraphics[width=\linewidth, height=0.52\textheight,
                       keepaspectratio]{assets/#1}
    \end{column}
    \begin{column}{0.35\textwidth}
      {\scriptsize #3}
    \end{column}
  \end{columns}%
}

% --- 1. System overview: block diagram -------------------------------------

\newcommand{\systemOverviewSlide}{%
\begin{frame}{System overview}
  \begin{center}
  \begin{tikzpicture}[node distance=4mm and 6mm, font=\scriptsize]
    % --- client block
    \node[block, minimum width=30mm, minimum height=15mm] (client)
      {CLIENT\\[1mm]\tiny replays the recording,\\assembles 64 KiB blocks};
    % --- server block
    \node[block, right=34mm of client, minimum width=36mm, minimum height=15mm] (server)
      {SERVER\\[1mm]\tiny validates, sorts on a worker pool,\\filters};
    % --- protocol band
    \node[accent block, above=16mm of $(client)!0.5!(server)$, minimum width=52mm]
      (proto) {protocol.h — one message definition};
    % --- flow arrows
    \draw[flow] ([yshift=3mm]client.east) -- node[edge label, above]
      {submit \{header, block\}} ([yshift=3mm]server.west);
    \draw[flow] ([yshift=-3mm]server.west) -- node[edge label, below]
      {ack} ([yshift=-3mm]client.east);
    \draw[flow dashed] ([yshift=-9mm]server.west) -- node[edge label, below]
      {pulse — async} ([yshift=-9mm]client.east);
    % --- protocol connects both
    \draw[flow dashed] (proto.south west) -- (client.north);
    \draw[flow dashed] (proto.south east) -- (server.north);
  \end{tikzpicture}
  \end{center}
  \vskip2mm
  \begin{highlight}
    Both programs are compiled against the same header, and the 64 KiB block
    crosses as the second part of an IOV — the header is read, validated, and
    only then is the payload pulled.
  \end{highlight}
\end{frame}%
}

% --- 3. Server flowchart ---------------------------------------------------

\newcommand{\serverFlowchartSlide}{%
\begin{frame}{Server: accept, validate, answer}
  \begin{center}
  \begin{tikzpicture}[node distance=3mm and 6mm, font=\tiny, scale=0.92,
                      transform shape, every node/.style={outer sep=1pt}]
    % main loop on the left, jobs down the right
    \node[accent block] (start) {\texttt{name\_attach("sortmed")}};
    \node[block, below=of start] (recv) {receive a message};
    \node[block, below=of recv] (type) {which type?};
    \node[block, right=12mm of type] (fetch) {\texttt{FETCH}\\reply the result};
    \node[block, left=14mm of type] (invalid) {refuse and reply};
    \node[block, below=of type] (check) {descriptor valid?\\checksum matches?};
    \node[block, below=of check] (queue) {reply the ack, then queue the job};
    \node[block, below=of queue] (sort) {workers: merge sort};
    \node[block, below=of sort] (median) {rolling median on the copy};
    \node[accent block, below=of median] (pulse) {\texttt{MsgSendPulse} — code 16};

    \draw[flow] (start) -- (recv);
    \draw[flow] (recv) -- (type);
    \draw[flow] (type) -- node[edge label, above]{fetch} (fetch);
    \draw[flow] (type) -- node[edge label, above]{refused} (invalid);
    \draw[flow] (type) -- node[edge label, right]{submit} (check);
    \draw[flow] (check) -- node[edge label, right]{yes} (queue);
    \draw[flow] (queue) -- (sort);
    \draw[flow] (sort) -- (median);
    \draw[flow] (median) -- (pulse);

    % the loop back to receiving, on whichever side a branch sits
    \draw[flow] (fetch.east) -- ++(5mm,0) |- node[edge label, right, pos=0.8]{} (recv.east);
    \draw[flow] (invalid.west) -- ++(-5mm,0) |- (recv.west);
    \draw[flow] (check.west) -- ++(-6mm,0) |- node[edge label, left, pos=0.75]{no} (recv.west);
    \draw[flow] (pulse.east) -- ++(5mm,0) |- (recv.east);
  \end{tikzpicture}
  \end{center}
\end{frame}%

% --- 4. Data flow ----------------------------------------------------------

\newcommand{\dataFlowSlide}{%
\begin{frame}{Data flow: what changes at each step}
  \begin{center}
  \begin{tikzpicture}[node distance=4mm and 4mm, font=\tiny, scale=0.95,
                      transform shape]
    \node[block, minimum width=26mm] (file)
      {recording on disk\\\tiny float32, 60\,000 samples\\one channel per file};
    \node[block, right=9mm of file, minimum width=29mm] (sensor)
      {virtual sensor\\\tiny 128-sample frames\\index $\rightarrow$ timestamp};
    \node[block, right=9mm of sensor, minimum width=29mm] (loader)
      {block loader\\\tiny 16\,384 samples = 64 KiB\\min, max, FNV-1a};
    \node[block, right=9mm of loader, minimum width=31mm] (wire)
      {message\\\tiny \{descriptor, block\}\\+ version and job id};

    \draw[flow] (file) -- node[edge label, above]{read} (sensor);
    \draw[flow] (sensor) -- node[edge label, above]{frame} (loader);
    \draw[flow] (loader) -- node[edge label, above]{submit} (wire);

    \node[block, below=12mm of sensor, minimum width=29mm] (sorted)
      {sorted copy\\\tiny order destroyed\\median as a by-product};
    \node[block, right=9mm of sorted, minimum width=29mm] (filtered)
      {time-ordered copy\\\tiny order intact\\rolling median};
    \node[accent block, right=9mm of filtered, minimum width=31mm] (result)
      {result\\\tiny median, window width,\\outliers, offset + checksum};

    \draw[flow dashed] (wire.south) -- ++(0,-5mm) -| (sorted.north);
    \draw[flow] (sorted) -- (filtered);
    \draw[flow] (filtered) -- (result);
  \end{tikzpicture}
  \end{center}
  \vskip2mm
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{highlight}
        Two buffers, not one. The sort destroys the time order the filter
        reads, so the arriving block is never sorted in place — its checksum
        describes it as it arrived.
      \end{highlight}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tech}
        Every hand-off in this chain is byte-deterministic: fixed-width
        fields, cleared structs, a zero-padded short block — so any two
        stages can be compared exactly.
      \end{tech}
    \end{column}
  \end{columns}
\end{frame}%
}
