% ---------------------------------------------------------------------------
% theme-b — Indigo & Violet (academic)
%
% Centred title page inside a ruled frame, a numbered section page, and a
% serif face for the prose under sans headings. Slower and more formal than
% theme A; suits a viva or a report defence.
% ---------------------------------------------------------------------------

\usepackage{mathpazo}                     % serif for body text

% --- palette ---------------------------------------------------------------

\definecolor{rtmPrimary}{HTML}{2B2D6E}   % indigo
\definecolor{rtmAccent}{HTML}{7C5CBF}    % violet
\definecolor{rtmAlert}{HTML}{C4553A}     % brick
\definecolor{rtmGood}{HTML}{2E7D6B}      % deep green
\definecolor{rtmInk}{HTML}{24272B}       % body text
\definecolor{rtmMuted}{HTML}{707886}     % captions, footer
\definecolor{rtmLine}{HTML}{C9CCD6}      % hairlines
\definecolor{rtmSurface}{HTML}{ECEAF4}   % box fills, code background

% --- beamer colours and fonts ----------------------------------------------

\setbeamercolor{normal text}{fg=rtmInk, bg=white}
\setbeamercolor{structure}{fg=rtmPrimary}
\setbeamercolor{frametitle}{fg=rtmPrimary, bg=white}
\setbeamercolor{title}{fg=rtmPrimary}
\setbeamercolor{subtitle}{fg=rtmMuted}
\setbeamercolor{author}{fg=rtmInk}
\setbeamercolor{date}{fg=rtmMuted}
\setbeamercolor{itemize item}{fg=rtmAccent}
\setbeamercolor{itemize subitem}{fg=rtmMuted}
\setbeamercolor{enumerate item}{fg=rtmAccent}
\setbeamercolor{alerted text}{fg=rtmAccent}
\setbeamercolor{footline}{fg=rtmMuted}
\setbeamercolor{block title}{fg=white, bg=rtmPrimary}
\setbeamercolor{block body}{fg=rtmInk, bg=rtmSurface}
\setbeamercolor{block title alerted}{fg=white, bg=rtmAlert}
\setbeamercolor{block body alerted}{fg=rtmInk, bg=rtmSurface}
\setbeamercolor{block title example}{fg=white, bg=rtmGood}
\setbeamercolor{block body example}{fg=rtmInk, bg=rtmSurface}

\setbeamerfont{frametitle}{size=\large, series=\bfseries}
\setbeamerfont{title}{size=\LARGE, series=\bfseries}
\setbeamerfont{subtitle}{size=\large}
\setbeamerfont{author}{size=\small}
\setbeamerfont{footline}{size=\tiny}
\setbeamerfont{caption}{size=\scriptsize}

% --- frame chrome ----------------------------------------------------------

\useinnertheme{default}
\useoutertheme{default}
\setbeamertemplate{navigation symbols}{}
\setbeamersize{text margin left=8mm, text margin right=8mm}

% Number the frame, then the title, with a full-width hairline under it.
\setbeamertemplate{frametitle}{%
  \vskip3mm%
  {\usebeamerfont{frametitle}%
   {\color{rtmAccent}\bfseries\insertframenumber\,}\hskip0.6em%
   \usebeamercolor[fg]{frametitle}\insertframetitle\par}%
  \vskip1.8mm%
  {\color{rtmLine}\hrule height 0.5pt}%
  \vskip1.5mm%
}

\setbeamertemplate{footline}{%
  \hbox to\paperwidth{%
    \hskip8mm%
    {\usebeamerfont{footline}\usebeamercolor[fg]{footline}\insertshortauthor}%
    \hfill%
    {\usebeamerfont{footline}\usebeamercolor[fg]{footline}%
      \insertshorttitle\hskip1em\textbar\hskip1em%
      \insertframenumber\,/\,\inserttotalframenumber}%
    \hskip8mm%
  }%
  \vskip2.2mm%
}

% --- title page ------------------------------------------------------------

\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture, overlay]
    % top rule only: the bottom of a title slide is left open
    \fill[rtmPrimary]
      ([yshift=-10mm]current page.north west) rectangle ([yshift=-11mm]current page.north east);
    \fill[rtmAccent]
      ([yshift=-11mm]current page.north west) rectangle ([yshift=-11.8mm]current page.north east);
    \node[anchor=center, inner sep=0pt, text width=0.80\paperwidth, align=center]
      at ([yshift=12mm]current page.center) {%
        {\color{rtmAccent}\footnotesize\bfseries\insertinstitute}\par
        \vskip5mm
        {\color{rtmPrimary}\LARGE\bfseries\inserttitle}\par
        \vskip4mm
        {\color{rtmMuted}\large\insertsubtitle}\par
        \vskip7mm
        {\color{rtmPrimary}\hrule height 0.6pt width 40mm}\par
      };
    \node[anchor=south, inner sep=0pt, text width=0.80\paperwidth, align=center]
      at ([yshift=16mm]current page.south) {%
        {\color{rtmInk}\small\authorsBlock}\par
        \vskip3mm
        {\color{rtmMuted}\small\insertdate}\par
      };
  \end{tikzpicture}%
}

% --- section pages ---------------------------------------------------------

\AtBeginSection[]{%
  \begin{frame}[plain]
    \vfill
    \begin{center}
      {\color{rtmAccent!35!white}\fontsize{52}{58}\selectfont\bfseries
       \insertsectionnumber}\par
      \vskip2mm
      {\color{rtmAccent}\hrule height 1pt width 24mm}\par
      \vskip4mm
      {\color{rtmPrimary}\LARGE\bfseries\insertsectionhead}\par
    \end{center}
    \vfill
  \end{frame}%
}

% --- tables, code, diagrams, macros ----------------------------------------
