% ---------------------------------------------------------------------------
% theme-c — Crimson & Amber (bold)
%
% Colour on every slide: a crimson band across the top of each frame, amber
% hairlines, and a full-bleed panel for the section and closing slides. The
% loudest of the four; good for a room, harder to read as dense notes.
% ---------------------------------------------------------------------------

% --- palette ---------------------------------------------------------------

\definecolor{rtmPrimary}{HTML}{8E1B2E}   % crimson
\definecolor{rtmAccent}{HTML}{C87A1C}    % amber
\definecolor{rtmAlert}{HTML}{B5442A}     % burnt orange
\definecolor{rtmGood}{HTML}{2F6B4F}      % deep green
\definecolor{rtmInk}{HTML}{2A2321}       % body text (warm near-black)
\definecolor{rtmMuted}{HTML}{8A7B76}     % captions, footer
\definecolor{rtmLine}{HTML}{E0D6D0}      % hairlines
\definecolor{rtmSurface}{HTML}{F5EFE9}   % box fills, code background

% --- beamer colours and fonts ----------------------------------------------

\setbeamercolor{normal text}{fg=rtmInk, bg=white}
\setbeamercolor{structure}{fg=rtmPrimary}
\setbeamercolor{frametitle}{fg=white, bg=rtmPrimary}
\setbeamercolor{title}{fg=rtmPrimary}
\setbeamercolor{subtitle}{fg=rtmMuted}
\setbeamercolor{author}{fg=rtmInk}
\setbeamercolor{date}{fg=rtmMuted}
\setbeamercolor{itemize item}{fg=rtmPrimary}
\setbeamercolor{itemize subitem}{fg=rtmMuted}
\setbeamercolor{enumerate item}{fg=rtmPrimary}
\setbeamercolor{alerted text}{fg=rtmAccent!85!black}
\setbeamercolor{footline}{fg=rtmMuted}
\setbeamercolor{block title}{fg=white, bg=rtmPrimary}
\setbeamercolor{block body}{fg=rtmInk, bg=rtmSurface}
\setbeamercolor{block title alerted}{fg=white, bg=rtmAlert}
\setbeamercolor{block body alerted}{fg=rtmInk, bg=rtmSurface}
\setbeamercolor{block title example}{fg=white, bg=rtmGood}
\setbeamercolor{block body example}{fg=rtmInk, bg=rtmSurface}

\setbeamerfont{frametitle}{size=\large, series=\bfseries}
\setbeamerfont{title}{size=\LARGE, series=\bfseries}
\setbeamerfont{subtitle}{size=\large}
\setbeamerfont{author}{size=\small}
\setbeamerfont{footline}{size=\tiny}
\setbeamerfont{caption}{size=\scriptsize}

% --- frame chrome ----------------------------------------------------------

\useinnertheme{default}
\useoutertheme{default}
\setbeamertemplate{navigation symbols}{}
\setbeamersize{text margin left=8mm, text margin right=8mm}

% Full-width crimson band, amber hairline under it.
\setbeamertemplate{frametitle}{%
  \nointerlineskip%
  \begin{beamercolorbox}[wd=\paperwidth, ht=3.4ex, dp=1.6ex,
                         leftskip=8mm, rightskip=8mm]{frametitle}%
    \usebeamerfont{frametitle}\insertframetitle%
  \end{beamercolorbox}%
  \vskip-0.4pt%
  {\color{rtmAccent}\hrule height 1.1pt}%
  \vskip1.6mm%
}

\setbeamertemplate{footline}{%
  {\color{rtmLine}\hrule height 0.4pt}%
  \hbox to\paperwidth{%
    \hskip8mm%
    {\usebeamerfont{footline}\usebeamercolor[fg]{footline}%
      \insertshortauthor\hskip1em\textbar\hskip1em\insertshorttitle}%
    \hfill%
    {\usebeamerfont{footline}%
      {\color{rtmPrimary}\bfseries\insertframenumber}%
      {\color{rtmMuted}\,/\,\inserttotalframenumber}}%
    \hskip8mm%
  }%
  \vskip2.2mm%
}

% --- title page ------------------------------------------------------------

\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[rtmPrimary]
      (current page.north west) rectangle ([yshift=-8mm]current page.north east);
    \fill[rtmAccent]
      ([yshift=-8mm]current page.north west) rectangle ([yshift=-9mm]current page.north east);
    \node[anchor=north west, inner sep=0pt, text width=0.84\paperwidth]
      at ([xshift=8mm, yshift=-24mm]current page.north west) {%
        {\color{rtmPrimary}\footnotesize\bfseries\insertinstitute}\par
        \vskip5mm
        {\color{rtmPrimary}\LARGE\bfseries\inserttitle}\par
        \vskip4mm
        {\color{rtmAccent}\hrule height 1pt width 22mm}\par
        \vskip4mm
        {\color{rtmMuted}\large\insertsubtitle}\par
      };
    \node[anchor=south west, inner sep=0pt, text width=0.84\paperwidth]
      at ([xshift=8mm, yshift=16mm]current page.south west) {%
        {\color{rtmInk}\small\authorsBlock}\par
        \vskip2mm
        {\color{rtmMuted}\small\insertdate}\par
      };
    \fill[rtmPrimary]
      ([yshift=8mm]current page.south west) rectangle ([yshift=6.8mm]current page.south east);
  \end{tikzpicture}%
}

% --- section pages ---------------------------------------------------------

\AtBeginSection[]{%
  \begin{frame}[plain]
    \begin{tikzpicture}[remember picture, overlay]
      \fill[rtmPrimary] (current page.south west) rectangle (current page.north east);
      \fill[rtmAccent]
        ([yshift=-6mm]current page.north west) rectangle ([yshift=-7.2mm]current page.north east);
      \node[anchor=west, inner sep=0pt, text width=0.84\paperwidth]
        at ([xshift=8mm]current page.center) {%
          {\color{rtmAccent}\footnotesize\bfseries PART \insertsectionnumber}\par
          \vskip3mm
          {\color{white}\LARGE\bfseries\insertsectionhead}\par
        };
    \end{tikzpicture}%
  \end{frame}%
}

\newcommand{\closingSlide}{%
  \begin{frame}[plain]
    \begin{tikzpicture}[remember picture, overlay]
      \fill[rtmPrimary]
        (current page.south west) rectangle ([yshift=-0.58\paperheight]current page.north east);
      \fill[rtmAccent]
        ([yshift=-0.58\paperheight]current page.north west)
        rectangle ([yshift=-0.592\paperheight]current page.north east);
      \node[anchor=north west, inner sep=0pt, text width=0.84\paperwidth]
        at ([xshift=8mm, yshift=-18mm]current page.north west) {%
          {\color{rtmAccent}\footnotesize\bfseries THANK YOU}\par
          \vskip5mm
          {\color{white}\Large\bfseries\inserttitle}\par
        };
      \node[anchor=south west, inner sep=0pt, text width=0.84\paperwidth]
        at ([xshift=8mm, yshift=12mm]current page.south west) {%
          {\color{rtmInk}\small\authOne}\par
          {\color{rtmInk}\small\authTwo}\par
          {\color{rtmInk}\small\authThree}\par
          \vskip2mm
          {\color{rtmMuted}\scriptsize\insertsubtitle}\par
        };
    \end{tikzpicture}%
  \end{frame}%
}
