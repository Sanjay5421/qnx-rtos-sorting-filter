% ---------------------------------------------------------------------------
% theme-d — Graphite & Lime (technical)
%
% Monospaced frame titles numbered like code listings, square-cornered boxes
% and diagrams, and a single lime accent against near-neutral greys. Reads as
% an engineering log rather than a lecture; the plainest of the four.
% ---------------------------------------------------------------------------

% --- palette ---------------------------------------------------------------

\definecolor{rtmPrimary}{HTML}{23272B}   % graphite
\definecolor{rtmAccent}{HTML}{5FA800}    % lime
\definecolor{rtmAlert}{HTML}{C9761B}     % amber
\definecolor{rtmGood}{HTML}{2E7D5B}      % green
\definecolor{rtmInk}{HTML}{23272B}       % body text
\definecolor{rtmMuted}{HTML}{7A8288}     % captions, footer
\definecolor{rtmLine}{HTML}{DDE1E3}      % hairlines
\definecolor{rtmSurface}{HTML}{F1F3F4}   % box fills, code background

% --- beamer colours and fonts ----------------------------------------------

\setbeamercolor{normal text}{fg=rtmInk, bg=white}
\setbeamercolor{structure}{fg=rtmPrimary}
\setbeamercolor{frametitle}{fg=rtmPrimary, bg=white}
\setbeamercolor{title}{fg=rtmPrimary}
\setbeamercolor{subtitle}{fg=rtmMuted}
\setbeamercolor{author}{fg=rtmInk}
\setbeamercolor{date}{fg=rtmMuted}
\setbeamercolor{itemize item}{fg=rtmAccent}
\setbeamercolor{itemize subitem}{fg=rtmMuted}
\setbeamercolor{enumerate item}{fg=rtmAccent}
\setbeamercolor{alerted text}{fg=rtmAccent!80!black}
\setbeamercolor{footline}{fg=rtmMuted}
\setbeamercolor{block title}{fg=white, bg=rtmPrimary}
\setbeamercolor{block body}{fg=rtmInk, bg=rtmSurface}
\setbeamercolor{block title alerted}{fg=white, bg=rtmAlert}
\setbeamercolor{block body alerted}{fg=rtmInk, bg=rtmSurface}
\setbeamercolor{block title example}{fg=white, bg=rtmGood}
\setbeamercolor{block body example}{fg=rtmInk, bg=rtmSurface}

% Monospaced titles: the one theme that changes a typeface.
\setbeamerfont{frametitle}{family=\ttfamily, size=\large, series=\bfseries}
\setbeamerfont{title}{size=\LARGE, series=\bfseries}
\setbeamerfont{subtitle}{size=\large}
\setbeamerfont{author}{size=\small}
\setbeamerfont{footline}{family=\ttfamily, size=\tiny}
\setbeamerfont{caption}{size=\scriptsize}

% --- frame chrome ----------------------------------------------------------

\useinnertheme{default}
\useoutertheme{default}
\setbeamertemplate{navigation symbols}{}
\setbeamersize{text margin left=8mm, text margin right=8mm}

% "d.03" — frame number in lime, title in monospace.
\setbeamertemplate{frametitle}{%
  \vskip3mm%
  {\usebeamerfont{frametitle}%
   {\color{rtmAccent}\bfseries d.\insertframenumber}\hskip1.4em%
   \usebeamercolor[fg]{frametitle}\insertframetitle\par}%
  \vskip1.8mm%
  {\color{rtmLine}\hrule height 0.4pt}%
  \vskip1.5mm%
}

\setbeamertemplate{footline}{%
  {\color{rtmLine}\hrule height 0.4pt}%
  \hbox to\paperwidth{%
    \hskip8mm%
    {\usebeamerfont{footline}\usebeamercolor[fg]{footline}%
      \insertshortauthor\hskip1.5em\insertshorttitle}%
    \hfill%
    {\usebeamerfont{footline}\usebeamercolor[fg]{footline}%
      [\insertframenumber/\inserttotalframenumber]}%
    \hskip8mm%
  }%
  \vskip2.2mm%
}

% --- title page ------------------------------------------------------------

\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[rtmPrimary]
      (current page.north west) rectangle ([yshift=-6mm]current page.north east);
    \fill[rtmAccent]
      ([yshift=-6mm]current page.north west) rectangle ([yshift=-7.4mm]current page.north east);
    \node[anchor=north west, inner sep=0pt, text width=0.84\paperwidth]
      at ([xshift=8mm, yshift=-22mm]current page.north west) {%
        {\color{rtmMuted}\ttfamily\footnotesize\bfseries\insertinstitute}\par
        \vskip5mm
        {\color{rtmPrimary}\LARGE\bfseries\inserttitle}\par
        \vskip4mm
        {\color{rtmAccent}\hrule height 1.6pt width 22mm}\par
        \vskip4mm
        {\color{rtmMuted}\large\insertsubtitle}\par
      };
    \node[anchor=south west, inner sep=0pt, text width=0.84\paperwidth]
      at ([xshift=8mm, yshift=14mm]current page.south west) {%
        {\color{rtmInk}\ttfamily\small \authorsBlock}\par
        \vskip2mm
        {\color{rtmMuted}\ttfamily\small\insertdate}\par
      };
  \end{tikzpicture}%
}

% --- section pages ---------------------------------------------------------

\AtBeginSection[]{%
  \begin{frame}[plain]
    \vfill
    {\color{rtmAccent}\bfseries\ttfamily PART \insertsectionnumber}\par
    \vskip2mm
    {\color{rtmPrimary}\LARGE\bfseries\insertsectionhead}\par
    \vskip3mm
    {\color{rtmLine}\hrule height 0.6pt}\par
    \vfill
  \end{frame}%
}

% --- tables, code, diagrams, macros ----------------------------------------

\setbeamertemplate{caption}[numbered]

  language=C,
  basicstyle=\ttfamily\scriptsize,
  keywordstyle=\color{rtmAccent!80!black}\bfseries,
  commentstyle=\color{rtmMuted}\itshape,
  stringstyle=\color{rtmPrimary},
  numberstyle=\ttfamily\tiny\color{rtmMuted},
  backgroundcolor=\color{rtmSurface},

  muted block/.style={block, draw=rtmMuted, fill=rtmSurface, text=rtmMuted},

}
