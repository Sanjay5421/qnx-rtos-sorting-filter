\section{Introduction}
% ===========================================================================

\begin{frame}{The Problem Statement}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.54\textwidth}
      A client offloads continuous \concept{floating-point time-series} sensor data to a server.
      \vskip1.5mm
      The server must:
      \begin{itemize}
        \item Sort blocks with a \concept{multi-threaded parallel merge sort},
        \item Filter sensor outliers with a \concept{rolling median filter},
        \item Notify completion asynchronously via a \concept{QNX pulse event}.
      \end{itemize}
      \vskip1.5mm
      The client then collects the calculated median and a memory-offset descriptor.
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{highlight}
        \textbf{Real-Time Constraints} \\
        High-rate telemetry arrives continuously. Impulsive sensor glitches distort statistical averages and must be removed without stalling collection.
      \end{highlight}
      \vskip1.5mm
      \begin{tech}
        \textbf{Architectural Rules} \\
        QNX synchronous IPC for submission, 2-part IOV transfer ($\ge$\,64 KiB payload), and an asynchronous completion pulse.
      \end{tech}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Why a Median Filter Over an Average?}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.54\textwidth}
      Sensor faults produce high-magnitude impulse spikes. A single faulty sample ruins an arithmetic mean completely:
      \vskip1mm
      \begin{center}
      \fitwidth{%
      \begin{tabular}{@{}lccc@{}}
        \toprule
        \rtmtablehead{Window Readings} & \rtmtablehead{Mean} & \rtmtablehead{Median} & \rtmtablehead{Truth} \\
        \midrule
        5.0, 5.0, 5.0, 5.0, 5.0   & 5.0 V  & 5.0 V & 5.0 V \\
        5.0, 5.0, 5.0, 5.0, 50.0  & \num{14.0 V} & \num{5.0 V} & 5.0 V \\
        \bottomrule
      \end{tabular}}
      \end{center}
      \vskip1mm
      A 50 V transient corrupts the mean by 9 V, but the median stays at 5 V.
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{itemize}
        \item \textbf{Odd kernel width}: Guarantees an unambiguous central sample.
        \item \textbf{Preserving time order}: Median filtering requires chronological order; it must run on a raw copy, not the sorted array.
        \item \textbf{Dual purpose}: Parallel sort gives block percentiles; rolling median cleans live signals.
      \end{itemize}
      \vskip1mm
      \begin{highlight}
        Rolling median cleanly strips $10$--$40\sigma$ transient spikes without distorting baseline data.
      \end{highlight}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Core Design Principles}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{tech}
        \textbf{Formal Contract First} \\
        A shared header (\texttt{protocol.h}) establishes message types, struct layouts, and 64 KiB block boundaries before writing application logic.
      \end{tech}
      \vskip1.5mm
      \begin{tech}
        \textbf{Modular Pipeline} \\
        Virtual sensor $\rightarrow$ block loader $\rightarrow$ message passing $\rightarrow$ parallel merge sort. Each module is fully testable on PC and target.
      \end{tech}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \concept{Zero padding overhead}: 16\,384 float32 samples is exactly 65\,536 bytes (64 KiB).
        \item \concept{Deterministic execution}: Stable merge logic guarantees identical results across 1, 2, 4, or 8 threads.
        \item \concept{Non-blocking server}: Synchronous ack releases the client immediately; processing continues in worker threads.
        \item \concept{Empirical rigor}: All performance numbers are measured directly on Raspberry Pi 4 hardware.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Virtual Sensor: Architecture \& Mechanics}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.51\textwidth}
      \texttt{app/sensor.c} emulates a hardware telemetry device:
      \smalltext\tighten
      \vskip1mm
      \compactitem
      \begin{itemize}
        \item \textbf{Production Interleaved Stream}: Reads directly from the interleaved telemetry record (\texttt{stride 4, offset k}) with sample formula:
        \vskip0.5mm
        {\scriptsize\texttt{idx = (i - win\_first) * 4 + offset}}
        \item \textbf{Sliding Window Buffer}: Keeps 4096 records in RAM (\texttt{SENSOR\_WINDOW\_SAMPLES}). Refills on demand via \texttt{fread()} as reads advance.
        \item \textbf{Device-Style Frames}: \texttt{sensor\_read()} delivers at most one frame (default 128 samples).
        \item \textbf{Synthetic Clock}: Exact sample arithmetic: $t = i \times 10^9 / f_{\text{rate}}$ ns.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{tech}
        \textbf{Strict Input Validation} \\
        Validates byte sizes upon opening. Explicitly rejects non-multiple float sizes, zero rates, mismatched record strides, or invalid channel offsets.
      \end{tech}
      \vskip1.5mm
      \begin{highlight}
        \textbf{Pacing Mechanism} \\
        \texttt{sensor\_read\_paced()} compares virtual delivery deadlines against monotonic clock (\texttt{clock\_gettime} / \texttt{nanosleep}) to hold real-world sensor rates.
      \end{highlight}
    \end{column}
  \end{columns}
\end{frame}

while (n < cap) {
    if (s->window_count == 0 ||

        \item \textbf{Zero Memory Bloat}: The full file never resides in memory at once.
        \item \textbf{Exact Stride}: Extracts the chosen channel with fixed stride 4 directly from the interleaved record.
      \end{itemize}

\begin{frame}{Block Loader: Aggregation \& Serialization}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.51\textwidth}
      \texttt{app/block\_loader.c} packages sensor frames into transport parcels:
      \vskip1mm
      \begin{itemize}
        \item \textbf{Exact 64 KiB Payload}: Reads frames until \texttt{PROTO\_BLOCK\_SAMPLES} (16\,384 floats = 65\,536 B) is reached.
        \item \textbf{Interleaved Channel Mapping}: Isolates field ordering in a single function:
        \vskip0.5mm
        {\scriptsize\texttt{Voltage $\rightarrow$ 1, Current $\rightarrow$ 2, Temp $\rightarrow$ 3}}
        \item \textbf{Zero-Padded Final Block}: Tail samples of incomplete blocks are explicitly zeroed so uninitialized memory is never transmitted.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{tech}
        \textbf{Descriptor Metadata} \\
        Generates \texttt{proto\_block\_desc\_t} for every block:
        \begin{itemize}
          \item Sequence number \& first sample index
          \item Valid sample count
          \item Min/Max boundaries
          \item 32-bit FNV-1a checksum
        \end{itemize}
      \end{tech}
      \vskip1.5mm
      \begin{highlight}
        Checksum covers \textbf{only declared samples}, not the zero-padding. Guarantees deterministic seals across all runs.
      \end{highlight}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Parallel Merge Sort: Phase Architecture}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.51\textwidth}
      \texttt{app/mergesort.c} coordinates a POSIX thread pool:
      \smalltext
      \vskip1mm
      \compactitem
      \begin{itemize}
        \item \textbf{Thresholding}: Inputs $<$\,4096 samples run serial merge sort (pool overhead exceeds benefit).
        \item \textbf{Phase 0 (Partitions)}: 64 KiB block splits into $k$ slices ($k = \min(n/1024, \text{threads})$). Workers sort slices concurrently into scratch.
        \item \textbf{Phases $1\dots \log_2(k)$ (Merges)}: Workers merge adjacent pairs of sorted runs. Buffer pointers swap (ping-pong) between phases.
        \item \textbf{Odd Phase Copy}: If final merge landed in scratch buffer, it is copied back to source in one \texttt{memcpy()}.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{highlight}
        \textbf{POSIX Barrier Synchronization} \\
        Uses 1 mutex and 2 condition variables:
        \begin{itemize}
          \item \texttt{work\_ready}: Broadcasts tasks.
          \item \texttt{phase\_done}: Signals completion.
        \end{itemize}
        No sleeps, no busy-waiting.
      \end{highlight}
      \vskip1.5mm
      \begin{tech}
        \textbf{Deterministic and Stable} \\
        Ties pick from the left run ($a_i \le b_j$). Equal items preserve order; output is bit-identical across 1, 2, 4, and 8 threads.
      \end{tech}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}[fragile]{Parallel Merge Sort: Barrier Code Walkthrough}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.54\textwidth}
\begin{lstlisting}
/* Post tasks to workers & wait for phase completion */
static void post_phase(mergesort_pool_t *p, unsigned ntasks) {
    pthread_mutex_lock(&p->lock);
    p->ntasks     = ntasks;
    p->next_task  = 0;
    p->tasks_done = 0;
    pthread_cond_broadcast(&p->work_ready);

    while (p->tasks_done < p->ntasks)
        pthread_cond_wait(&p->phase_done, &p->lock);

    pthread_mutex_unlock(&p->lock);
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{itemize}
        \item \textbf{Manager Thread}: Prepares task list, broadcasts \texttt{work\_ready}, blocks on \texttt{phase\_done}.
        \item \textbf{Worker Threads}: Pull next available task index under mutex, execute sort/merge, increment \texttt{tasks\_done}.
        \item \textbf{Last Finisher}: Signals \texttt{phase\_done} when \texttt{tasks\_done == ntasks}.
      \end{itemize}
      \vskip1mm
      \key{Strict synchronization without artificial delays}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Format Clarification: Planar vs Interleaved}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{tech}
        \textbf{Planar Format (Single Channel File)} \\
        One flat file per signal (e.g. \texttt{r1\_voltage.bin}). 60\,000 contiguous floats (240\,000 B).
        \vskip0.8mm
        $\text{sample}_i = \text{file}[i]$ (\texttt{stride 1, offset 0}).
      \end{tech}
      \vskip1.5mm
      \begin{tech}
        \textbf{Interleaved Format (Production Payload)} \\
        One unified record file (\texttt{Chen2020\_c1...bin}). 60\,000 frames $\times$ 4 float32 (960\,000 B).
        \vskip0.8mm
        $\text{sample}_i = \text{file}[4i + k]$ (\texttt{stride 4, offset k}).
      \end{tech}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{highlight}
        \textbf{Why the Distinction Matters in Code} \\
        Sorting planar data directly is meaningful. On an interleaved file, sorting without selecting channel offset $k$ sorts seconds against volts against amperes!
      \end{highlight}
      \vskip1.5mm
      \begin{itemize}
        \item \texttt{block\_loader\_channel\_offset()} isolates this knowledge (Voltage is offset 1, not 2).
        \item Planar was used only to cross-check offsets during initial testing.
        \item \concept{Interleaved is our sole production format}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
